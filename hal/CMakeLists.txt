# Works with 3.14 and tested through 3.28
cmake_minimum_required(VERSION 3.14...3.28)

# Project name and a few useful settings. Other commands can pick up the results
project(
  hal
  VERSION 0.1
  DESCRIPTION "FORMOSA Hardware Abstraction Layer"
  LANGUAGES CXX C)

install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
add_subdirectory(src)

option(HAL_ENABLE_TESTING "Build tests for HAL" OFF)
if(HAL_ENABLE_TESTING)
  message(STATUS "Enabling testing for HAL")
  add_subdirectory(tests)
endif()

option(HAL_ENABLE_GCOV "Enable gcov support for HAL" OFF)
if(HAL_ENABLE_GCOV)
  message(STATUS "Enabling gcov support for HAL")
endif()

# Build gpufw_elf_carr
find_program(XXD_EXECUTABLE xxd REQUIRED)
find_program(OBJCOPY_EXECUTABLE NAMES riscv64-none-elf-objcopy llvm-objcopy objcopy REQUIRED)
set(GPUFW_BIN "${CMAKE_CURRENT_BINARY_DIR}/gpufw.bin")
set(GPUFW_ELF_C "${CMAKE_CURRENT_BINARY_DIR}/gpufw.elf.c")
set(GPUFW_ELF_H "${CMAKE_CURRENT_BINARY_DIR}/gpufw.elf.h")
set(GPUFW_GEN_NAME "gpufw_elf")
set(GPUFW_ELF_INC_CONTENT
"/* This file is generated by CMake */
#pragma once

extern unsigned char ${GPUFW_GEN_NAME}[];
extern unsigned int ${GPUFW_GEN_NAME}_len;
"
)

file(WRITE ${GPUFW_ELF_H} "${GPUFW_ELF_INC_CONTENT}")

add_custom_command(
  OUTPUT ${GPUFW_BIN}
  COMMAND ${OBJCOPY_EXECUTABLE} -O binary "$<TARGET_FILE:gpufw.elf>" ${GPUFW_BIN}
  DEPENDS gpufw.elf
  COMMENT "Extracting ${GPUFW_BIN}..."
)

add_custom_command(
  OUTPUT ${GPUFW_ELF_C}
  COMMAND ${XXD_EXECUTABLE} -i -n ${GPUFW_GEN_NAME} ${GPUFW_BIN} ${GPUFW_ELF_C}
  DEPENDS ${GPUFW_BIN}
  COMMENT "Embedding ${GPUFW_ELF_C}..."
)

add_library(gpufw_elf_carr OBJECT
  ${GPUFW_ELF_C}
)

target_include_directories(gpufw_elf_carr
  PUBLIC "${CMAKE_CURRENT_BINARY_DIR}"
)
